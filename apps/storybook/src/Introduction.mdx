<DocSection n={2} title="Install" meta="npm" description="Tokens alone work with any framework. The components need React 19.">

```bash
npm install @med-o/design-tokens @med-o/ui   # tokens + React components
```

```tsx

import '@med-o/ui/styles.css';
```

```tsx
// src/main.tsx – once, at the root of the app
import { createRoot } from 'react-dom/client';
import { I18nProvider } from 'react-aria-components';
import '@fontsource/dm-sans/400.css';
import '@fontsource/dm-sans/500.css';
import '@fontsource/dm-sans/600.css';
import '@fontsource/dm-sans/700.css';
import '@fontsource/dm-mono/400.css';
import '@med-o/design-tokens/css/tokens.css';
import '@med-o/design-tokens/css/typography.css';
import '@med-o/ui/styles.css';
import { App } from './App';

// The locale formats dates and numbers and picks the built-in texts. Same language as <html lang>.
createRoot(document.getElementById('root')!).render(
  <I18nProvider locale="en-US">
    <App />
  </I18nProvider>,
);
```

```tsx
// src/App.tsx
import { Button, Container, Stack, TextField } from '@med-o/ui';

}
```

```tsx
// app/providers.tsx – React context needs a client component
'use client';
import { I18nProvider } from 'react-aria-components';

export function Providers({ children }: { children: React.ReactNode }) {
  // The locale formats dates and numbers and picks the built-in texts. Same language as <html lang>.
  return <I18nProvider locale="en-US">{children}</I18nProvider>;
}
```

```tsx
// app/layout.tsx – global styles and fonts are imported once, here
import '@fontsource/dm-sans/400.css';
import '@fontsource/dm-sans/500.css';
import '@fontsource/dm-sans/600.css';
import '@fontsource/dm-sans/700.css';
import '@fontsource/dm-mono/400.css';
import '@med-o/design-tokens/css/tokens.css';
import '@med-o/design-tokens/css/typography.css';
import '@med-o/ui/styles.css';
import { Providers } from './providers';

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en">
      <body>
        <Providers>{children}</Providers>
      </body>
    </html>
  );
}
```

```tsx
// app/page.tsx – event handlers such as onSubmit need a client component
'use client';
import { Button, Container, Stack, TextField } from '@med-o/ui';

export default function Page() {
  return (
    <Container as="main" size="prose">
      <Stack as="form" gap="fields" onSubmit={(e) => e.preventDefault()}>
        <h1 className="medo-text-heading-lg">Sign up</h1>
        <TextField
          label="Email"
          type="email"
          isRequired
          errorMessage="Enter a valid email address, e.g. name@example.com."
        />
        <Button type="submit">Create account</Button>
      </Stack>
    </Container>
  );
}
```

Components without event handlers can be used in server components as well – the package is marked `"use client"`.

<DocSection n={4} title="Use the tokens" meta="CSS · Tailwind" description="Import the variables once. Then use only semantic tokens.">

```css
/* main.css */
@import "@med-o/design-tokens/css/tokens.css";    /* variables, light + dark */
@import "@med-o/design-tokens/css/typography.css"; /* optional: .medo-text-* classes */

.card {
  background: var(--medo-color-surface-raised);
  color: var(--medo-color-text-primary);
  padding: var(--medo-space-6);
  border-radius: var(--medo-radius-lg);
  box-shadow: var(--medo-shadow-elevation-2);
}
```
